/** * Envelope budget — email relay * * Watches your inbox for bank transaction alerts, pulls the amount and * merchant out of each one, and holds them for the budget app to fetch. * Replaces the whole Shortcut approach: every charge lands in the tray a * few minutes after it happens, tap or swipe or online or subscription. * * SETUP — see the walkthrough in chat, short version: * 1. script.google.com -> New project, paste this over Code.gs * 2. Edit SECRET and SENDER below * 3. Run install() once, approve the permission prompts * 4. Run testParse() and read the log to check it's reading your alerts * 5. Deploy -> New deployment -> Web app, Execute as Me, Access Anyone * 6. Paste the /exec URL into the app under Sync link */ var SECRET = '6EDDLqCHnQsWMpWL0Q1S'; // any random string you invent var SENDER = 'capitalone.com'; // add others: 'capitalone.com OR yourbank.com' var LABEL = 'budget-synced'; var KEEP = 300; // charges retained /** Run once. Creates the label and the every-5-minutes trigger. */ function install() { if (!GmailApp.getUserLabelByName(LABEL)) GmailApp.createLabel(LABEL); ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'scanInbox') ScriptApp.deleteTrigger(t); }); ScriptApp.newTrigger('scanInbox').timeBased().everyMinutes(5).create(); Logger.log('Installed. Label and 5-minute trigger are set.'); } /** Dry run — logs what it would pull, saves nothing. */ function testParse() { var threads = GmailApp.search('from:(' + SENDER + ') newer_than:7d', 0, 10); if (!threads.length) return Logger.log('No alerts found. Check SENDER, and that alerts are on.'); threads.forEach(function (th) { th.getMessages().forEach(function (m) { var hit = parseAlert(m); Logger.log(hit ? ('OK ' + hit.amount + ' ' + hit.who) : ('MISS ' + m.getSubject())); }); }); } /** The scheduled job. */ function scanInbox() { var label = GmailApp.getUserLabelByName(LABEL); if (!label) return; var store = PropertiesService.getScriptProperties(); var list = JSON.parse(store.getProperty('charges') || '[]'); var seen = {}; list.forEach(function (c) { seen[c.rid] = 1; }); GmailApp.search('from:(' + SENDER + ') newer_than:3d -label:' + LABEL, 0, 25) .forEach(function (th) { th.getMessages().forEach(function (m) { var id = m.getId(); if (seen[id]) return; var hit = parseAlert(m); if (!hit) return; list.push({ rid: id, who: hit.who, amount: hit.amount, ts: m.getDate().getTime() }); seen[id] = 1; }); th.addLabel(label); }); if (list.length > KEEP) list = list.slice(list.length - KEEP); store.setProperty('charges', JSON.stringify(list)); } /** Debug: dump the wording of the newest real alert so the parser can be matched to it. */ function showAlert() { var th = GmailApp.search('from:(' + SENDER + ') subject:(transaction OR charged OR purchase) newer_than:14d', 0, 1); if (!th.length) return Logger.log('No transaction alert found in the last 14 days.'); var m = th[0].getMessages().pop(); Logger.log('SUBJECT: ' + m.getSubject()); Logger.log('BODY: ' + m.getPlainBody().replace(/\s+/g, ' ').slice(0, 700)); } /** Pull an amount and a merchant out of one alert. Adjust if your bank rewords. */ function parseAlert(msg) { var subject = msg.getSubject() || ''; var text = subject + '\n' + msg.getPlainBody(); text = text.replace(/\s+/g, ' '); // marketing is judged on the SUBJECT only - footers are full of // words like "offer" and "rewards" and were killing real alerts if (/credit score|credit usage|line of credit|pre-?approved|bonus|refer|offer|rewards|statement is ready/i.test(subject)) return null; if (/payment (received|posted)|thank you for your payment|autopay/i.test(text)) return null; if (!/(transaction|purchase|charged|was used|debit|withdrawal)/i.test(text)) return null; var a = text.match(/\$\s?([\d,]+\.\d{2})/); if (!a) return null; var amount = parseFloat(a[1].replace(/,/g, '')); if (!amount) return null; var who = ''; var pats = [ /\bat\s+([A-Za-z0-9][A-Za-z0-9 &'.\-*#]{2,40})/, /\bto\s+([A-Za-z0-9][A-Za-z0-9 &'.\-*#]{2,40})/, /merchant[:\s]+([A-Za-z0-9][A-Za-z0-9 &'.\-*#]{2,40})/i ]; for (var i = 0; i < pats.length && !who; i++) { var m = text.match(pats[i]); if (m) who = m[1]; } who = who .replace(/\b(on|was|has|for|using|with|your|the|account|card|ending).*$/i, '') .replace(/\s{2,}/g, ' ') .trim(); if (who.length < 3) who = 'Purchase'; if (who === who.toUpperCase()) { who = who.toLowerCase().replace(/\b[a-z]/g, function (c) { return c.toUpperCase(); }); } return { amount: Math.round(amount * 100) / 100, who: who.slice(0, 40) }; } /** What the app calls. */ function doGet(e) { var p = (e && e.parameter) || {}; var store = PropertiesService.getScriptProperties(); var list = JSON.parse(store.getProperty('charges') || '[]'); if (p.add) { // manual write, kept for the Shortcut route if (p.key !== SECRET) return out({ error: 'bad key' }); var amt = Math.abs(parseFloat(String(p.amt || '').replace(/[^0-9.]/g, ''))); if (!amt) return out({ error: 'no amount' }); list.push({ rid: Utilities.getUuid(), who: String(p.add).slice(0, 40), amount: Math.round(amt * 100) / 100, ts: Date.now() }); if (list.length > KEEP) list = list.slice(list.length - KEEP); store.setProperty('charges', JSON.stringify(list)); return out({ ok: true, held: list.length }); } // reads require the key too - the URL alone must not be enough if (p.key !== SECRET) return out({ error: 'bad key' }); var since = Number(p.since || 0); // the app only advances `since` after it has saved a charge, so anything // at or below it is confirmed on the phone - drop it here rather than // letting a history pile up in Google's storage if (since > 0) { var kept = list.filter(function (c) { return c.ts > since; }); if (kept.length !== list.length) store.setProperty('charges', JSON.stringify(kept)); list = kept; } return out(list); } function out(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
